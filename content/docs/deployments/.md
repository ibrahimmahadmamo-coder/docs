import aiohttp, asyncio, re, json, time
from bs4 import BeautifulSoup
import dns.resolver
from ..bot.config import DYNAMIC_DISCOVERY_INTERVAL

class APIDiscoverer:
    def __init__(self, redis_client):
        self.redis = redis_client
        self.discovered_endpoints = set()
        
    async def start(self):
        while True:
            await self.scan_for_new_dexes()
            await self.scan_for_new_cex_endpoints()
            await self.benchmark_latency()
            await asyncio.sleep(DYNAMIC_DISCOVERY_INTERVAL)
    
    async def scan_for_new_dexes(self):
        """Crawl DeFi Llama, CoinGecko, DEXScreener for new DEXs"""
        sources = [
            "https://api.llama.fi/protocols",
            "https://api.coingecko.com/api/v3/coins/markets?vs_currency=usd"
        ]
        async with aiohttp.ClientSession() as session:
            for url in sources:
                try:
                    async with session.get(url, timeout=5) as resp:
                        data = await resp.json()
                        # Extract new DEX addresses and endpoints
                        # Store in Redis with TTL
                        self.redis.sadd("discovered_dexs", str(data)[:500])
                except Exception as e:
                    print(f"Discovery error: {e}")
    
    async def scan_for_new_cex_endpoints(self):
        """Discover new CEX API endpoints via DNS/service discovery"""
        cex_domains = ["binance.com", "kraken.com", "coinbase.com", "kucoin.com", "bybit.com", "okx.com"]
        for domain in cex_domains:
            try:
                # Find API subdomains
                answers = dns.resolver.resolve(f"api.{domain}", 'A')
                for rdata in answers:
                    endpoint = f"https://api.{domain}"
                    self.redis.sadd("cex_endpoints", endpoint)
            except:
                pass
    
    async def benchmark_latency(self):
        """Test each discovered endpoint for speed – keep only fastest"""
        all_endpoints = list(self.redis.smembers("cex_endpoints")) + list(self.redis.smembers("discovered_dexs"))
        for endpoint in all_endpoints:
            start = time.time()
            try:
                async with aiohttp.ClientSession() as session:
                    async with session.get(endpoint, timeout=2) as resp:
                        latency_ms = (time.time() - start) * 1000
                        self.redis.hset("endpoint_latency", endpoint, latency_ms)
            except:
                self.redis.hset("endpoint_latency", endpoint, 9999)

import asyncio, random, time, numpy as np
from sklearn.linear_model import LinearRegression
from .config import ENABLE_PREDICTION

class AdaptiveHunters:
    def __init__(self, bot, redis_client):
        self.bot = bot
        self.redis = redis_client
        self.model = LinearRegression() if ENABLE_PREDICTION else None
        self.history = []
        
    async def start(self):
        tasks = [
            self.auto_discover_and_hunt(),
            self.learn_profit_patterns(),
            self.rotate_strategies()
        ]
        await asyncio.gather(*tasks)
    
    async def auto_discover_and_hunt(self):
        """Hunts using ONLY currently discovered fast endpoints"""
        while self.bot.running:
            # Get top 3 fastest endpoints
            latencies = self.redis.hgetall("endpoint_latency")
            if not latencies:
                await asyncio.sleep(1)
                continue
            fastest = sorted(latencies.items(), key=lambda x: float(x[1]))[:3]
            
            for endpoint, _ in fastest:
                # Check for arbitrage on this endpoint
                profit = await self.check_arbitrage(endpoint)
                if profit > 0:
                    await self.bot.execute_trade(profit, f"adaptive_{endpoint[:20]}")
            await asyncio.sleep(0.1)  # 10ms loop – hyper speed
    
    async def check_arbitrage(self, endpoint):
        # Simplified – would query endpoint for prices
        return random.uniform(0.5, 5.0) if random.random() < 0.1 else 0
    
    async def learn_profit_patterns(self):
        """ML model predicts best times to hunt"""
        if not self.model:
            return
        while self.bot.running:
            # Collect recent profit data
            self.history.append({
                'time': time.time(),
                'profit': self.bot.total_profit
            })
            if len(self.history) > 100:
                X = np.array([[h['time'] % 86400] for h in self.history[-100:]])
                y = np.array([h['profit'] for h in self.history[-100:]])
                self.model.fit(X, y)
                # Predict next best hour
                next_hour = np.array([[ (time.time() + 3600) % 86400 ]])
                prediction = self.model.predict(next_hour)
                self.redis.set("predicted_profit_next_hour", float(prediction[0]))
            await asyncio.sleep(60)
    
    async def rotate_strategies(self):
        """Dynamically drop failing strategies, add new ones"""
        strategy_scores = {}
        while self.bot.running:
            for strategy in ['cex_arb', 'dex_arb', 'mev', 'liquidation']:
                # Get last hour profit per strategy from Redis
                score = float(self.redis.get(f"strategy_score_{strategy}") or 0)
                strategy_scores[strategy] = score
            # Drop worst strategy if underperforming
            if strategy_scores and min(strategy_scores.values()) < 1.0:
                worst = min(strategy_scores, key=strategy_scores.get)
                print(f"🗑️ Dropping underperforming strategy: {worst}")
                # Disable it (set flag in Redis)
                self.redis.set(f"strategy_enabled_{worst}", "0")
            await asyncio.sleep(300)

import sqlite3, time, aiohttp, asyncio
from .config import DB_PATH, TELEGRAM_BOT_TOKEN, TELEGRAM_CHAT_ID

def init_db():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS trades
                 (id INTEGER PRIMARY KEY, timestamp REAL, source TEXT, profit_usd REAL)''')
    c.execute('''CREATE TABLE IF NOT EXISTS withdrawals
                 (id INTEGER PRIMARY KEY, timestamp REAL, amount_usd REAL, tx_hash TEXT, destination TEXT)''')
    conn.commit()
    conn.close()

def get_total_profit():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("SELECT COALESCE(SUM(profit_usd),0) FROM trades")
    total = c.fetchone()[0]
    conn.close()
    return total

def add_trade(source, profit_usd):
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("INSERT INTO trades (timestamp,source,profit_usd) VALUES (?,?,?)",
              (time.time(), source, profit_usd))
    conn.commit()
    conn.close()

def add_withdrawal(amount_usd, tx_hash, destination):
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute("INSERT INTO withdrawals (timestamp,amount_usd,tx_hash,destination) VALUES (?,?,?,?)",
              (time.time(), amount_usd, tx_hash, destination))
    conn.commit()
    conn.close()
    asyncio.create_task(send_telegram(f"💰 WITHDRAWAL: ${amount_usd:.2f} to {destination}"))
    
async def send_telegram(msg):
    if TELEGRAM_BOT_TOKEN and TELEGRAM_CHAT_ID:
        url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
        async with aiohttp.ClientSession() as session:
            await session.post(url, json={"chat_id": TELEGRAM_CHAT_ID, "text": msg})

def stats():
    conn = sqlite3.connect("data/beast.db")
    c = conn.cursor()
    c.execute("SELECT COUNT(*) FROM trades")
    trade_count = c.fetchone()[0]
    c.execute("SELECT COALESCE(SUM(amount_usd),0) FROM withdrawals")
    withdrawn = c.fetchone()[0]
    conn.close()
    # Get adaptive stats
    fastest = redis_client.hgetall("endpoint_latency")
    fastest = sorted(fastest.items(), key=lambda x: float(x[1]))[:3] if fastest else []
    return jsonify({
        'profit': sum([float(r) for r in redis_client.lrange("profit_history", -100, -1)]),

        'withdrawn': withdrawn,
        'fastest_endpoints': [f"{e[:30]}: {l}ms" for e,l in fastest]
    })

def start():
    global bot, thread
    if bot is None:
        bot = ApexPredator(socketio)
        thread = threading.Thread(target=lambda: asyncio.run(bot.start()))
